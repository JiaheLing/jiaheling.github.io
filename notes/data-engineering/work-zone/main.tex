% =======================
% Custom chapter page
%
% Does not print the default:
% “第一章”
%
% The complete chapter title is supplied manually.
% Example:
% \chapterpage{第1章：计算机网络}
% =======================
\newcommand{\chapterpage}[1]{%
  \clearpage
  \refstepcounter{chapter}%
  \addcontentsline{toc}{chapter}{#1}%
  \thispagestyle{empty}%

  % Vertical position
  \vspace*{1.2in}

  % Horizontal position
  \hspace*{0.2in}

  {\Huge\bfseries #1\par}

  \clearpage
  \markboth{#1}{#1}%
}

% Lecture header

% }

% \lecture{计算机网络概述}{}
% =======================
\newcommand{\lecture}[2]{%
  \section{#1}%
  \IfStrEq{#2}{}{%
    % No summary box
  }{%
    \begin{summarybox}
      #2
    \end{summarybox}
  }%
}
